===========================
Instalacja Backend / Slim 4
===========================

# 1. Struktura projektu
D:\_projekty\mapalotow

# 2. Utwórz katalog backend

cd /d/_projekty/mapalotow
mkdir backend
cd backend

# 3. Sprawdź PHP
php -v

# 4. Sprawdź Composer
composer --version

5. Utwórz composer.json
composer init

# 6. Instalujemy Slim 4
composer require slim/slim:"4.*"

# następnie implementację PSR-7:
composer require slim/psr7

# 7. Dodaj obsługę .env
composer require vlucas/phpdotenv

# 8. Utwórz katalog public
mkdir public

# I utwórz:
# D:\_projekty\mapalotow\backend\public\index.php

<?php

declare(strict_types=1);

use Psr\Http\Message\ResponseInterface as Response;
use Psr\Http\Message\ServerRequestInterface as Request;
use Slim\Factory\AppFactory;

require __DIR__ . '/../vendor/autoload.php';

$app = AppFactory::create();

$app->addRoutingMiddleware();

$app->addErrorMiddleware(
    true,
    true,
    true
);

$app->get('/api/health', function (
    Request $request,
    Response $response
): Response {

    $data = [
        'status' => 'ok',
        'application' => 'Mapa Lotow API',
    ];

    $response->getBody()->write(
        json_encode(
            $data,
            JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES
        )
    );

    return $response
        ->withHeader('Content-Type', 'application/json');
});

$app->run();
